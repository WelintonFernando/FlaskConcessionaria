create database oficina;
use oficina;

-- Tabela cliente
create table cliente (
    idcliente int primary key auto_increment,
    nome varchar(60) not null,
    endereco varchar(100),
    cidade varchar(80),
    uf varchar(2),
    cep varchar(9)
);

-- Tabela prestador
create table prestador (
    idprestador int primary key auto_increment,
    nome_empresa varchar(60) not null,
    cidade varchar(80),
    uf varchar(2),
    cep varchar(9)
);

-- Tabela veiculo
create table veiculo (
    idplaca varchar(9) primary key,
    ano int,
    modelo int,
    preco_fipe decimal(10,2),
    fabricante varchar(50),
    modelo_veiculo varchar(60),
    cor varchar(20),
    preco_venda decimal(10,2),
    total_despesa decimal(10,2)
);

-- Tabela compra
create table compra (
    idcompra int primary key auto_increment,
    idplaca varchar(9) not null,
    idcliente int not null,
    data date,
    valor_pago decimal(10,2),
    forma_pagamento varchar(40),
    foreign key (idplaca) references veiculo(idplaca),
    foreign key (idcliente) references cliente(idcliente)
);

-- Tabela venda
create table venda (
    idvenda int primary key auto_increment,
    data date,
    valor_vendido decimal(10,2),
    idcliente int not null,
    idplaca varchar(9) not null,
    forma_pagamento varchar(40),
    foreign key (idcliente) references cliente(idcliente),
    foreign key (idplaca) references veiculo(idplaca)
);

-- Tabela despesa
create table despesa (
    iddespesa int primary key auto_increment,
    idplaca varchar(9) not null,
    descricao varchar(80),
    valor decimal(10,2),
    idprestador int not null,
    data_servico date,
    foreign key (idplaca) references veiculo(idplaca),
    foreign key (idprestador) references prestador(idprestador)
);
